    {

        WriteTimestampedLogEntry("LOOT: donor miss for type " + IntToString(nType)

            + ", " + IntToString(nAll) + " properties; type " + IntToString(nType)
            + " sub:cost =" + (sSeen == "" ? " none" : sSeen));

    itemproperty ipNone;
    return ipNone;
